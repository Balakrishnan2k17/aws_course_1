name: Build, Push & Deploy Docker Image

on:
  push:
    branches:
      - aws-cicd
  workflow_dispatch:

permissions:
  id-token: write
  contents: read

env:
  AWS_REGION: us-east-2
  ECR_IMAGE: 262647287021.dkr.ecr.us-east-2.amazonaws.com/rt-etrm-backend-aws:latest
  ECS_CLUSTER: rt-etrm
  ECS_SERVICE: etrm-backend-aws-service-7ncq0vak

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:

    - name: Checkout Code
      uses: actions/checkout@v4

    - name: Configure AWS Credentials (OIDC)
      uses: aws-actions/configure-aws-credentials@v4
      with:
        role-to-assume: arn:aws:iam::262647287021:role/GitHubActionsECSDeployRole
        aws-region: ${{ env.AWS_REGION }}

    - name: Login to ECR
      uses: aws-actions/amazon-ecr-login@v2

    - name: Build Docker Image (No Cache)
      run: |
        docker build --no-cache \
        -t $ECR_IMAGE \
        -f ./services/Dockerfile \
        ./services

    - name: Push Docker Image
      run: |
        docker push $ECR_IMAGE

    - name: Wait for Image Availability
      run: sleep 15

    - name: Force New Deployment
      run: |
        aws ecs update-service \
          --cluster $ECS_CLUSTER \
          --service $ECS_SERVICE \
          --force-new-deployment

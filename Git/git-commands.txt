Git: Version Control System

{
git init
git add README.md
git commit -m "first commit"
git branch -M main
git remote add origin https://github.com/Balakrishnan2k17/file.git
git push -u origin main
}

git clone https://github.com/<username>/<repo.git>

git status

main branch: --> create a new file

vim index.html
git add .
git status
git commit -m "Added index.html file"
git push origin main


another branch: --> merge request

git branch dev
git branch
git checkout dev
git branch
git add . 
git commit -m "msg"
git push origin dev


merge dev to main branch --> merge different branch

git checkout main
git pull origin main
git merge dev
git push origin main


main branch: --> another changes

git checkout main
vim index.html
git commit -m "Added index.html file v2"
git push origin main


main branch: --> rollback the changes

git log --oneline
git revert a7e9a3e
git push origin main

create one workflow for CICD.

paste the code:

name: Deploy via SCP & SSH

on:
  push:
    branches:
      - dev

env:
  IMAGE_NAME: cicd_for_aws
  CONTAINER: cicd_for_aws

jobs:
  deploy:
    runs-on: ubuntu-latest

    steps:

    - name: Checkout Code
      uses: actions/checkout@v4

    - name: Build Docker Image
      run: docker build -t $IMAGE_NAME .

    - name: Save Docker Image
      run: docker save $IMAGE_NAME > cicd_for_aws.tar

    - name: Create SSH Key File
      run: |
        echo "${{ secrets.EC2_SSH_KEY }}" | base64 -d > key.pem
        chmod 600 key.pem

    - name: Copy Image to EC2 (SCP)
      run: |
        scp -o StrictHostKeyChecking=no -i key.pem cicd_for_aws.tar ubuntu@54.91.4.222:/home/ubuntu/

    - name: Deploy on EC2 via SSH
      run: |
        ssh -o StrictHostKeyChecking=no -i key.pem ubuntu@54.91.4.222 << EOF
        docker stop $CONTAINER || true
        docker rm $CONTAINER || true
        docker load < cicd_for_aws.tar
        docker run -d -p 8080:80 --name $CONTAINER $IMAGE_NAME
        EOF
